// Ground 改造：
function Ground() {
  const [tex, setTex] = useState<THREE.Texture | null>(null);
  useEffect(() => {
    const url = streetTileUrl('asphalt_main');
    if (!url) return;
    new THREE.TextureLoader().load(url, (t) => {
      t.wrapS = t.wrapT = THREE.RepeatWrapping;
      t.repeat.set(5, 5);
      t.colorSpace = THREE.SRGBColorSpace;
      setTex(t);
    });
  }, []);
  return (
    <mesh rotation={[-Math.PI/2, 0, 0]} position={[0, 0, 0]} receiveShadow>
      <planeGeometry args={[40, 40]} />
      <meshStandardMaterial map={tex ?? undefined} color={tex ? '#ffffff' : '#141a24'} roughness={0.95} />
    </mesh>
  );
}

// Roads 替换（line 62-89）：
import { Road } from './Road';
// 替换 <Roads /> 为 <RoadsLayer />：
function RoadsLayer() {
  return (
    <>
      {WEALTH_DISTRICTS.filter((d) => d.id !== 'finance').map((d) => {
        const c = districtCenter(d.id);
        const dx = -c.x, dz = -c.z;
        const len = Math.sqrt(dx * dx + dz * dz);
        const angle = Math.atan2(dx, dz);
        const mid = { x: c.x / 2, z: c.z / 2 };
        return <Road key={d.id} from={[c.x, c.z]} to={[0, 0]} kind={len > 14 ? 'main' : 'side'} />;
      })}
    </>
  );
}

import { useEffect, useState } from 'react';
import * as THREE from 'three';
import { propUrl } from '@/assets/images/wealth';

interface Props { x: number; z: number; rotation?: number; variant?: 'a' | 'b' | 'c'; }

export function StreetLight({ x, z, rotation = 0, variant = 'a' }: Props) {
  const url = propUrl('streetlamp', variant);
  const [tex, setTex] = useState<THREE.Texture | null>(null);
  useEffect(() => { /* TextureLoader */ }, [url]);

  return (
    <group position={[x, 0, z]} rotation={[0, rotation, 0]}>
      {/* 简化几何底座（缺失贴图时） */}
      <mesh castShadow position={[0, 0.02, 0]}>
        <cylinderGeometry args={[0.05, 0.08, 0.05, 8]} />

      </mesh>
      <mesh castShadow position={[0, 0.6, 0]}>
        <cylinderGeometry args={[0.04, 0.04, 1.2, 6]} />
        <meshStandardMaterial color="#888" metalness={0.6} />
      </mesh>
      {/* 灯头 */}
      <mesh castShadow position={[0, 1.2, 0]}>

      {/* 灯贴图 sprite（Billboard 朝相机） */}
      {tex && (
        <sprite position={[0, 1.25, 0]} scale={[0.4, 1, 1]}>

      )}

## 8. 依赖关系图

```
P0-A (docs)
    │
    ▼
P0-B (assets + index.ts helpers)
    │
    ├──► P1-A (Ground + Road) ──► P1-B (BuildingMesh)
    │                                    │
    │                                    └──► P1-C (props + layer)
    │
    └──► P2 (polish, optional)
```

**关键依赖**：
- P0-B → P1-A：需要 `streetTileUrl` helper。
- P0-B → P1-B：需要 `districtFacadeUrl` / `districtRoofUrl` helper。
- P0-B → P1-C：需要 `propUrl` helper。
- P1-A → P1-B：无强依赖（可并行）。
- P1-A → P1-C：StreetPropsLayer 可能引用 Road 的 `from/to`（共享路径数据）。
- P1-B → P1-C：BuildingMesh 不依赖 props。

**可优化并发**：P0-B 跑完资产后，P1-A / P1-B / P1-C 可由 3 个 SubAgent 并行实施（CLAUDE.md §13）。
